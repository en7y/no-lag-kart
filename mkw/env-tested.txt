# Mario Kart World 1.8.0: the settings the final build was measured with (mkw/README.md). They are the game profile's defaults (dm/games/0400C3F00006E000/game.tsv),
# so a build needs no env file; the bench (mkw/test) passes this file to every run, so each run states its settings.
# Lines that start with # are comments. A file of the same form at ~/Library/Application Support/nxnative/0400C3F00006E000/env.txt overrides the app's settings.

# Console (docked) mode for one to four players: the game draws for a 2560x1440 picture, with reflection maps of twice the size and a larger shadow map than in handheld mode.
# Split screen stays in console mode (no DM_OPMODE_MULTI), and the game's dynamic resolution reads its frames' real GPU time (no DM_NVN_GPU_TS_PAD): no setting here
# lowers what the game draws.
DM_OPMODE=1
# MetalFX scales the picture up to the window or the screen
DM_PRESENT_UPSCALE=metalfx
# A late frame does not lengthen the hold of the frames after it
DM_FPS_HOLD=off
# The GPU clock is kept at its top: in fullscreen with Game Mode on and the keeper off it averaged 1,199 MHz of 1,398. About 4.5 W more in a race: auto on battery
DM_GPU_KEEP=1
# A submission goes to the GPU in segments of 8 draws while the GPU has run out of work (default 64), so the GPU starts sooner after the game submits
DM_NVN_CUT=8
# Upload sources are copied into a 128 MB resident ring instead of buffers made resident for each upload: a scene cut's residency work halves
DM_NVN_UP_COPY=128
